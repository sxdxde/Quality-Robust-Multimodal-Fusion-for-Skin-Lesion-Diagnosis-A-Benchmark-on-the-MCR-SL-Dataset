\section{Motivation}
\begin{frame}{Why This Problem?}
\begin{itemize}
\item Deep learning has reached dermatologist-level accuracy on \textbf{image-only} skin lesion classification (Esteva et al., 2017; HAM10000).
\item But a real dermatologist never diagnoses from an image alone --- age, sex, lesion site, referral history all matter.
\vspace{0.3cm}
\item A smaller line of work fuses images with \textbf{patient metadata} (e.g. MetaBlock on PAD-UFES-20) and shows it helps.
\vspace{0.3cm}
\item \textbf{Missing piece:} none of these datasets record how \emph{good} the diagnostic image itself was.
\begin{itemize}
\item Real clinical images vary hugely in quality (lighting, focus, hair, blur).
\item No prior benchmark could ask: \emph{do errors concentrate on poor-quality images?}
\end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{The Opportunity: MCR-SL}
A brand-new dataset (Castro-Fernandez et al., \emph{Data}, 2025) closes this gap:
\begin{itemize}
\item Every diagnostic image has an expert-assigned \textbf{1--10 quality rating}.
\item Rich structured metadata per lesion/subject (22 fields).
\item Partial histopathology confirmation (ground-truth beyond panel opinion).
\end{itemize}
\vspace{0.4cm}
\begin{block}{Key fact}
\textbf{No prior published benchmark exists on MCR-SL.} We are first.
\end{block}
\vspace{0.2cm}
This lets us ask a question no earlier skin-lesion benchmark could:\\
\emph{Does model performance degrade on lower-quality images, and can we train the model to be more robust to that?}
\end{frame}

\section{Our Contributions}
\begin{frame}{What This Work Does}
\begin{enumerate}
\item First published benchmark on MCR-SL: image-only vs.\ late-fusion vs.\ \textbf{channel-gated fusion}, under a subject-disjoint 5-fold protocol.
\vspace{0.1cm}
\item \textbf{A quality-adaptive loss reweighting scheme} (\texttt{hard\_mining}) that uses MCR-SL's expert quality ratings to weight the training loss --- \alert{and a shuffled-quality control that validates the gain is genuinely quality-driven}, not generic reweighting.
\vspace{0.1cm}
\item \textbf{Six robustness analyses} enabled by MCR-SL's quality ratings, expert certainty, histopathology subset, and multi-lesion subjects.
\vspace{0.1cm}
\item \textbf{Three apparent positive findings that we overturned ourselves} using proper controls --- a validation discipline we report as part of the contribution, not an appendix.
\vspace{0.1cm}
\item Final result: \textbf{0.840 $\pm$ 0.095 balanced accuracy} (\texttt{hard\_mining} + TTA + multi-image averaging), \emph{within noise of} the closest verified binary comparator (0.836) using ${\sim}10\times$ less training data.
\end{enumerate}
\vspace{0.15cm}
{\footnotesize The architecture is unchanged from standard components. The contribution is the \emph{loss weighting} and the \emph{validation}, not a new network.}
\end{frame}

\begin{frame}{Why MCR-SL, Not a Bigger Dataset?}
\alert{The honest answer: it is not about scale. It is about a field that only exists here.}
\vspace{0.25cm}
\centering
\small
\begin{tabular}{|l|c|c|}
\hline
\textbf{Dataset} & \textbf{Size} & \textbf{Expert per-image quality rating?} \\
\hline
ISIC (2019/2020) & 25{,}000--35{,}000+ & \alert{No} \\
HAM10000 & 10{,}015 & \alert{No} \\
PAD-UFES-20 & 2{,}298 & \alert{No} \\
\rowcolor{green!12}\textbf{MCR-SL} & \textbf{240 lesions} & \textbf{Yes --- 1--10, per diagnostic image} \\
\hline
\end{tabular}
\vspace{0.35cm}

\raggedright
\normalsize
\begin{itemize}
\item Larger datasets buy \textbf{statistical power}. None of them can answer this project's research question, because \textbf{the field the mechanism depends on does not exist in them}.
\item MCR-SL's ratings are \textbf{real expert judgements of the photograph}, tied to the specific image used for each diagnosis --- not an automated blur score or a learned proxy.
\item We accept the cost of small $N$ (wide error bars, reported throughout) in exchange for asking a question that is \textbf{unanswerable elsewhere}.
\end{itemize}
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{Has Our Research Question Been Answered?}
The project asks \textbf{two separable questions}. They get two different answers.
\vspace{0.25cm}

\begin{block}{Q1: Does performance degrade on lower-quality images? --- \alert{No (null result)}}
Spearman(rating, error) $=-0.093$, $p=0.158$, $N=231$; tercile pattern non-monotonic (0.776 / 0.914 / 0.868). Reported as a \textbf{genuine null} --- neither confirmed nor ruled out. \textbf{Not hidden.}
\end{block}
\begin{block}{Q2: Can the model be trained to be more robust to quality? --- \textbf{Yes, with evidence}}
\texttt{hard\_mining} loss reweighting: \textbf{+0.037} balanced accuracy, \textbf{+0.092} sensitivity, \textbf{+0.025} AUROC.\\[2pt]
\textbf{Validated, not asserted:} a shuffled-quality control (same weight distribution, randomised lesion$\to$rating mapping) \textbf{collapses the entire gain}.
\end{block}
{\footnotesize Q1 being null does \emph{not} invalidate Q2 --- a model can be made more robust to a factor that shows no significant baseline correlation at this $N$.}
\end{frame}

\section{Architecture}
\begin{frame}{Architecture --- Two Encoders}
\begin{columns}[T]
\column{0.5\textwidth}
\textbf{Image Encoder}
\begin{itemize}
\item EfficientNet-B0, ImageNet-pretrained
\item Input: dermoscopic image, $224\times224$
\item Output: 1280-d pooled vector \emph{and} the last conv feature map ($1280\times7\times7$)
\end{itemize}
\column{0.5\textwidth}
\textbf{Metadata Encoder}
\begin{itemize}
\item 16 categorical fields $\to$ learned embeddings (with explicit ``unknown'' category --- \emph{no imputation})
\item 4 numeric fields (age, height, weight, diameter) $\to$ z-scored using \emph{train-fold-only} statistics + missingness bit
\item Concat $\to$ 2-layer MLP $\to$ 128-d metadata vector
\end{itemize}
\end{columns}
\vspace{0.4cm}
\centering
\small Both encoders feed into one of two fusion strategies (next slide).
\end{frame}

\begin{frame}{Architecture --- Two Fusion Strategies}
\begin{block}{1. Late fusion (baseline)}
concat(pooled image vector, metadata vector) $\to$ Linear+ReLU+Dropout (256-d) $\to$ classifier.\\
This is the standard MetaBlock-style approach from prior literature.
\end{block}
\vspace{0.3cm}
\begin{block}{2. Channel-gated fusion (our main method)}
metadata vector $\to$ Linear $\to$ Sigmoid $\to$ a 1280-d \textbf{gate} $\in [0,1]^{1280}$\\
$\to$ multiply elementwise against the image's conv feature map, channel-by-channel\\
$\to$ global pool $\to$ same 256-d projection $\to$ classifier.\\[4pt]
\emph{Idea:} let patient metadata decide \textbf{which visual channels matter}, instead of just concatenating two vectors. Same mechanism as Squeeze-and-Excitation, but the gating signal comes from metadata, not from the image itself.
\end{block}
\end{frame}

\begin{frame}{Architecture --- Three Prediction Heads}
All three heads sit on top of the \textbf{same} 256-d fused vector; only their outputs and losses differ.
\vspace{0.1cm}
\small
\begin{block}{1. Binary head (main task, always on)}
256-d $\to$ Linear $\to$ 1 logit $\to$ sigmoid $\to$ $P$(malignant). \\
Loss: weighted BCE (or focal loss), class weights recomputed per training fold ($\sim$1:4.6 malignant:non-malignant). \\
\emph{This is the only head whose output is reported as the paper's headline results.}
\end{block}
\begin{block}{2. Auxiliary 9-class head (exploratory)}
256-d $\to$ Linear $\to$ 9 logits $\to$ softmax $\to$ unified diagnosis (NEV, SK, BCC, \ldots). \\
Loss: class-weighted CE, downweighted to $\sim$0.3--0.5$\times$ the binary loss. \\
\emph{Reported only as an exploratory table --- several classes have $<$10 lesions.}
\end{block}
\begin{block}{3. Quality-regression head (optional, quality-aware variant only)}
256-d $\to$ Linear $\to$ 1 scalar $\to$ predicted mean expert quality rating (1--10). Loss: MSE, weighted 0.1--0.2$\times$. \\
\emph{Tests whether \textbf{predicting} quality improves robustness --- it did not. Later superseded by \textbf{weighting the loss} with quality instead, which did work.}
\end{block}
\end{frame}

\section{Training \& Evaluation}
\begin{frame}{Training / Evaluation Protocol}
\begin{block}{Subject-disjoint stratified 5-fold cross-validation}
\begin{itemize}
\item Split by \texttt{subject\_id}, never by lesion --- a subject can have multiple lesions, and images of the same lesion are near-duplicates.
\item Folds balanced by greedily equalizing malignant-lesion counts per fold.
\item For each rotation: 1 fold held out for \textbf{test}, 1 disjoint fold held out only for \textbf{checkpoint selection} (val), rest for training.
\item \textbf{No hyperparameter tuning against final test-fold results.}
\end{itemize}
\end{block}
\vspace{0.3cm}
\begin{block}{Losses}
\begin{itemize}
\item Binary head: weighted BCE, class weights recomputed per training fold ($\sim$1:4.6 malignant:non-malignant overall).
\item Auxiliary 9-class head: class-weighted CE, downweighted ($\sim$0.3--0.5$\times$), exploratory only.
\item (Optional) Quality-regression head: low weight (0.1--0.2), predicts mean expert rating.
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{The 5 Folds, Visualized}
60 subjects $\to$ 5 disjoint folds (balanced by malignant-lesion count). Every rotation uses a \textbf{different} test fold and a different, disjoint val fold --- so all 5 folds eventually serve as the test fold exactly once.
\vspace{0.3cm}
\centering
\begin{tabular}{c|ccccc}
\textbf{Rotation} & \textbf{Fold 0} & \textbf{Fold 1} & \textbf{Fold 2} & \textbf{Fold 3} & \textbf{Fold 4} \\
\hline
1 & \cellcolor{red!25}Test & \cellcolor{orange!35}Val & Train & Train & Train \\
2 & Train & \cellcolor{red!25}Test & \cellcolor{orange!35}Val & Train & Train \\
3 & Train & Train & \cellcolor{red!25}Test & \cellcolor{orange!35}Val & Train \\
4 & Train & Train & Train & \cellcolor{red!25}Test & \cellcolor{orange!35}Val \\
5 & \cellcolor{orange!35}Val & Train & Train & Train & \cellcolor{red!25}Test \\
\end{tabular}
\vspace{0.4cm}

\begin{itemize}
\item \textbf{Train} folds: model weights are updated here.
\item \textbf{Val} fold: used only to pick the best training checkpoint (never influences weights).
\item \textbf{Test} fold: touched exactly once, only for final reported metrics --- never used for any decision.
\end{itemize}
\vspace{0.2cm}
Final numbers = mean $\pm$ std of the 5 \textbf{test}-fold results (one per rotation).
\end{frame}

\begin{frame}{Evaluation Metrics --- What Each One Means}
\centering
\scriptsize
\begin{tabular}{|l|p{4.3cm}|p{4.7cm}|}
\hline
\textbf{Metric} & \textbf{Definition} & \textbf{Why it matters here} \\
\hline
Accuracy & (correct predictions) / (all predictions) & Can be misleading: predicting ``non-malignant'' always still scores $\sim$82\% \\
\hline
\textbf{Balanced accuracy} & average of sensitivity and specificity & \textbf{Primary metric} --- corrects for the malignant class being a minority ($\sim$18\%) \\
\hline

\hline

\hline
Specificity & (non-malignant cleared) / (all actual non-malignant) & Low specificity means unnecessary biopsies/patient anxiety \\
\hline
AUROC & probability the model ranks a random malignant lesion above a random non-malignant one & Threshold-independent measure of overall ranking quality \\
\hline
\end{tabular}
\vspace{0.3cm}

\normalsize All reported as mean $\pm$ std across the 5 test folds, plus a confusion matrix aggregated across all folds.
\end{frame}

\section{Quality-Adaptive Loss}
\begin{frame}{Quality-Adaptive Loss: The Mechanism}
Every lesion has a mean expert quality rating $q \in [1,10]$. We use it to \textbf{weight the training loss per sample} --- not to predict it as an output (that was the auxiliary head, which failed).
\vspace{0.25cm}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{block}{\texttt{trust} --- down-weight poor images}
\[ w = 0.5 + \tfrac{q-1}{9} \quad\Rightarrow\quad [1,10] \to [0.5,\,1.5] \]
\emph{``Low-quality images are less reliable, so trust them less.''}
\end{block}
\column{0.5\textwidth}
\begin{block}{\texttt{hard\_mining} --- up-weight poor images}
\[ w = 1.5 - \tfrac{q-1}{9} \quad\Rightarrow\quad [1,10] \to [1.5,\,0.5] \]
\emph{``Force the model to work harder on the difficult images.''}
\end{block}
\end{columns}
\vspace{0.3cm}
\centering
$\mathcal{L}_i = \underbrace{w^{\text{class}}_i}_{\text{existing class weight}} \times \underbrace{w^{\text{quality}}_i}_{\text{new}} \times \ \text{BCE}(\hat{y}_i, y_i)$
\vspace{0.25cm}

\raggedright
{\footnotesize Lesions with no valid rating get a neutral weight of $1.0$. Fixed range, two directions, \textbf{no grid search} --- the range was not tuned. Architecture, optimiser, folds and checkpoint rule are \textbf{identical} to the baseline; only the loss weighting changes.}
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{Validating the Loss: The Shuffled-Quality Control}
\begin{block}{The obvious objection}
``Maybe \emph{any} per-sample reweighting acts as a regulariser. Perhaps the quality ratings are irrelevant and you have just found noise that happens to help.''
\end{block}
\vspace{0.15cm}
\textbf{The control:} within each training fold, \textbf{shuffle} the quality ratings across lesions. Identical weight \emph{distribution}, but the lesion$\to$rating correspondence is destroyed. Test folds always use \textbf{real, unshuffled} ratings. Everything else identical.
\vspace{0.25cm}
\centering
\small
\begin{tabular}{|l|c|c|c|}
\hline
\textbf{Mechanism} & \textbf{Bal.\ acc.} & \textbf{Sens.} & \textbf{High$-$Low gap} \\
\hline
Plain baseline (no quality) & 0.783 & 0.672 & 0.091 \\
\hline
\rowcolor{green!12}\texttt{hard\_mining} (real ratings) & \textbf{0.820} & \textbf{0.764} & 0.091 \\
\rowcolor{red!10}\texttt{hard\_mining} (\alert{shuffled}) & 0.775 & 0.667 & 0.016 \\
\hline
\rowcolor{yellow!15}\texttt{trust} (real ratings) & 0.788 & 0.725 & 0.082 \\
\rowcolor{red!10}\texttt{trust} (\alert{shuffled}) & 0.763 & 0.672 & 0.021 \\
\hline
\end{tabular}
\vspace{0.3cm}

\raggedright
\normalsize
\begin{itemize}
\item \textbf{\texttt{hard\_mining} passes:} shuffling collapses it to 0.775 / 0.667 --- \emph{back to baseline}. The gain \textbf{requires the real quality signal}.
\item \textbf{\texttt{trust} fails:} its apparently-narrower tercile gap (0.082) is \emph{beaten} by the shuffled control (0.021). Revealed as a reweighting artifact --- reported as a negative result.
\end{itemize}
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{Where Does the Gain Actually Come From?}
Decomposing the final result --- \textbf{how much is genuinely about quality?}
\vspace{0.2cm}
\centering
\small
\begin{tabular}{|c|l|c|c|c|c|}
\hline
\textbf{Step} & \textbf{Configuration} & \textbf{Uses quality?} & \textbf{Bal.\ acc.} & \textbf{$\Delta$ vs base} & \textbf{Sens.} \\
\hline
0 & Baseline (channel-gated) & No & 0.783 & --- & 0.672 \\
\rowcolor{green!12}1 & + \texttt{hard\_mining} loss & \textbf{Yes} & 0.820 & \textbf{+0.037} & 0.764 \\
2 & + TTA (eval-time) & No & 0.833 & +0.050 & \textbf{0.808} \\
3 & + multi-image (eval-time) & No & \textbf{0.840} & +0.057 & 0.783 \\
\hline
\rowcolor{red!10}ctrl & \texttt{hard\_mining}, \alert{shuffled} & Fake & 0.775 & $-$0.008 & 0.667 \\
\hline
\end{tabular}
\vspace{0.35cm}

\raggedright
\normalsize
\begin{itemize}
\item Of the total \textbf{+0.057} gain, \textbf{+0.037 ($\approx$65\%)} is the validated, quality-specific contribution.

\item Steps 2--3 require \textbf{no retraining} at all.
\end{itemize}
\end{frame}

\item Beating a prior number was never really possible --- \textbf{no prior benchmark exists on MCR-SL}.

\section{Validation Discipline}
\begin{frame}{Three Times We Almost Reported a False Positive}
Each of these \textbf{looked like a result}. A control overturned it. We report all three.
\vspace{0.2cm}
\centering
\scriptsize
\begin{tabular}{|p{3.1cm}|p{3.3cm}|p{4.6cm}|}
\hline
\textbf{What it looked like} & \textbf{The check} & \textbf{What it actually was} \\
\hline
\texttt{trust} narrowed the quality-tercile gap $0.091 \to 0.082$ --- a second quality-robustness win
& Shuffled-quality control
& \alert{Reversed.} The \emph{shuffled} control produced an even narrower gap (\textbf{0.021}). A generic reweighting artifact, not a quality effect. \\
\hline
Expert diagnostic certainty predicted model error, $\rho=-0.175$, \textbf{$p=0.008$} --- apparently significant
& Stratify by malignant / non-malignant class
& \alert{Reversed.} Dissolves within both strata ($p=0.083$ / $p=0.559$). Malignant lesions cluster at low certainty, so the pooled correlation was \textbf{class-mix confounding}. \\
\hline
``Train on all images per lesion, not just one'' --- a large untapped data opportunity
& Verify the loader before building anything
& \alert{False premise.} Image-level training had been the default since run one ($5.65\times$ more samples than lesions). \textbf{Caught before any effort was wasted.} \\
\hline
\end{tabular}
\vspace{0.3cm}

\raggedright
\normalsize
At $N=234$ with $\sim$8--9 malignant lesions per fold, \textbf{an uncontrolled first look reliably produces plausible positives}. We treat catching them as part of the contribution --- and it is the most transferable lesson here.
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{What Is Our Novelty, Precisely?}
\alert{Not the architecture. Not a new loss function class. Not state of the art.}
\vspace{0.25cm}
\begin{enumerate}
\item \textbf{First use of \emph{real expert-assigned} quality labels for quality-adaptive loss reweighting in medical image classification.} The closest precedents --- \textbf{AdaFace} (CVPR 2022) and \textbf{MagFace} --- come from face recognition and must use \emph{feature-norm as a proxy}, because face datasets carry no expert quality annotation. MCR-SL does.
\vspace{0.1cm}
\item \textbf{The opposite weighting direction works here.} AdaFace/MagFace \emph{de-emphasise} low-quality samples; our \texttt{hard\_mining} \emph{up-weights} them, and that is the variant that wins.\\
{\small \emph{Plausible mechanism:} a dermatologist can often still diagnose correctly from a poor photograph, so the label remains valid and the hard case is worth learning from. In face ID, poor quality can make identity genuinely \textbf{unrecoverable} --- so de-emphasis is right there, and wrong here.}
\vspace{0.1cm}
\item \textbf{Distinguished from the nearest skin-lesion prior work.} \texttt{arXiv:2308.02119} (HAM10000) performs expert-informed sample reweighting, but weights by \textbf{how the label was confirmed} (diagnostic method). We weight by a \textbf{direct expert rating of the photograph itself} --- a different signal answering a different question.
\vspace{0.1cm}
\item \textbf{The validation discipline is part of the claim.} The shuffled-quality control, and the three reversed findings, are reported as contributions --- not buried.
\end{enumerate}
\end{frame}

\begin{frame}{Cross-Dataset Check: DeepDRiD (Not Completed)}
\small
\begin{block}{What we set out to test}
Does \texttt{hard\_mining} transfer to a \textbf{second medical imaging domain} with a differently-structured quality annotation? \textbf{DeepDRiD} (diabetic retinopathy, ISBI 2020) ships per-image quality annotations alongside DR severity grades --- the closest analogue to MCR-SL's ratings outside dermatology.
\end{block}
\begin{block}{Status: \alert{gate failed --- no results, and none estimated}}
A deliberate go/no-go gate ran \textbf{before} any training code was written, and it \textbf{failed}:
\begin{itemize}
\item The transfer was attempted and \textbf{aborted after 50 minutes} --- \texttt{RPC failed (result=92)}, \texttt{early EOF}, \texttt{index-pack failed}. All 2{,}916 objects compressed server-side; the connection could not sustain a single stream long enough to deliver the pack.
\item Measured throughput \textbf{40--64 KB/s} single-stream. Parallelism reaches ${\sim}700$ KB/s, but \texttt{git} is single-stream and the archive endpoint \textbf{ignores HTTP range requests}, so neither path can be parallelised. \texttt{git-lfs} is also unavailable.
\end{itemize}
\end{block}
\centering
\normalsize\textbf{No DeepDRiD number is reported, estimated, or implied anywhere in this deck.}\\
{\footnotesize Concrete future work. Having a gate that \emph{can} fail --- and honouring it --- is the same discipline as the three reversed findings.}
\end{frame}

\begin{frame}{Limitations}
\begin{itemize}
\item \alert{\textbf{Effect size vs.\ uncertainty:}} our fold-to-fold std ($\pm$0.078--0.095 balanced accuracy) is \textbf{larger than most differences we report}. The \texttt{hard\_mining} effect (+0.037) is supported by the shuffled control, but \textbf{no two configs within ${\sim}0.05$ should be called a win} over one another.
\item \textbf{Single seed}, single institution, 234 usable lesions, $\sim$8--9 malignant lesions per test fold --- one lesion flipping moves a fold's sensitivity by ${\sim}0.11$.
\item \textbf{Folds are subject-imbalanced} (5 / 6 / 16 / 16 / 16 subjects) --- balanced on malignant-lesion count, not on subject count.
\item \textbf{No external validation.} The DiffMIC comparison is cross-dataset context, not a benchmark. The DeepDRiD cross-domain check \textbf{was not completed}.
\item Ground truth mostly panel-consensus (28/240 histopathology-confirmed); several 9-class categories have $<$10 lesions --- exploratory only.
\item Quality terciles are modest ($N\approx$53--93); \textbf{six analyses were run with no multiple-comparison correction}.
\item Metadata-importance ranking is confounded by embedding-dimension imbalance (12-d categorical vs.\ 1 scalar numeric).
\item A known \texttt{pos\_weight} miscalibration ($\sim$16\%, lesion-level weight applied to image-level batches) is \textbf{quantified and disclosed}, not silently corrected.
\end{itemize}
\end{frame}

\begin{frame}{Conclusion \& Future Work}
\small
\begin{block}{What we found}
\begin{itemize}
\item \textbf{First benchmark on MCR-SL.} Final: \textbf{0.840 $\pm$ 0.095} balanced accuracy --- \emph{within noise} of the closest verified binary comparator, with ${\sim}10\times$ less data.
\item \textbf{Q1 (does quality degrade performance?): null} --- no significant relationship at $N=231$.
\item \textbf{Q2 (can we train for quality robustness?): yes} --- \texttt{hard\_mining} gives +0.037 balanced accuracy / +0.092 sensitivity, and \textbf{a shuffled control proves the gain depends on the real ratings}. Sensitivity on the \emph{worst} images rises $0.667 \to 0.810$.
\item \textbf{Three apparent positives we overturned ourselves}, plus quality-aware training, contrastive loss and alternative optimisers --- all reported as negatives.
\end{itemize}
\end{block}
\begin{block}{Future work}
Complete the \textbf{DeepDRiD} cross-domain check; \textbf{multi-seed averaging} to shrink error bars that currently exceed most effect sizes; correct the disclosed \texttt{pos\_weight} miscalibration; use the 779 unused clinical images with a modality indicator.
\end{block}
\end{frame}
